    function calculateROI() {
        const leads = parseInt(document.getElementById('leads-input').value);
        const value = parseInt(document.getElementById('value-input').value);
        
        document.getElementById('lead-val').innerText = leads;
        document.getElementById('value-val').innerText = value.toLocaleString();

         Calculations
        const currentConv = 0.03;  3% manual
        const aiConv = 0.08;       8% with AI response
        const hoursPerLead = 0.5;  30 mins saved per lead process

        const extraRevenue = (leads  aiConv  value) - (leads  currentConv  value);
        const hoursSaved = leads  hoursPerLead  12;

        document.getElementById('roi-hours').innerText = Math.round(hoursSaved);
        document.getElementById('roi-revenue').innerText = $ + (extraRevenue  1000).toFixed(1) + k;
    }
